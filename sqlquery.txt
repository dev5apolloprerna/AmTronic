-- Login rule + HSN (2026-09-21): only designations flagged `can_login`
-- (e.g. Sales) may log in. Same as running the two migrations
-- 2026_09_21_000006 and 2026_09_21_000007.
-- Run AFTER the "Masters update" block above. Use EITHER this file
-- OR `php artisan migrate`, not both.
-- (The HSN change 5-6 -> 4/6/8 digits needs no SQL: the columns already fit 8 digits.)
-- =====================================================================

ALTER TABLE `designations` ADD `can_login` tinyint(1) NOT NULL DEFAULT 0 AFTER `name`;

-- Nobody who can log in today may lose access:
-- 1) regular users with no designation are the sales staff -> "Sales" (created if missing)
INSERT INTO `designations` (`name`, `status`, `can_login`, `created_at`, `updated_at`)
SELECT 'Sales', 'active', 0, NOW(), NOW() FROM DUAL
WHERE EXISTS (SELECT 1 FROM `users` WHERE `role` = 'user' AND `designation_id` IS NULL)
  AND NOT EXISTS (SELECT 1 FROM `designations` WHERE `name` = 'Sales');

UPDATE `users`
SET `designation_id` = (SELECT `id` FROM `designations` WHERE `name` = 'Sales')
WHERE `role` = 'user' AND `designation_id` IS NULL
  AND EXISTS (SELECT 1 FROM `designations` WHERE `name` = 'Sales');

-- 2) any designation already held by a regular user keeps its holders' access
UPDATE `designations`
SET `can_login` = 1
WHERE `id` IN (SELECT DISTINCT `designation_id` FROM `users` WHERE `role` = 'user' AND `designation_id` IS NOT NULL);

-- Employees whose designation cannot log in need no email or password
-- (the unique index on email is kept; MySQL allows many NULLs in a unique column)
-- Length is 191, NOT 255: the app sets Schema::defaultStringLength(191), so that is what these
-- columns are. 255 would fail on servers with the 767-byte index limit (ERROR 1709/1071).
ALTER TABLE `users` MODIFY `email` varchar(191) NULL, MODIFY `password` varchar(191) NULL;
